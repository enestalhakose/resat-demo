Python ve e-posta entegrasyonu

Mevcut sisteminizdeki ürün listesi otomatik olarak ilanlara aktarılır, gelen talepler e-posta ile size düşer.

01Veri kaynağı

Excel, ERP veya veritabanınızdaki ürün listesi

02Python servisi

03API aktarımı

Gelen teklif ilgili yetkiliye iletilir

Örnek: toplu ilan aktarımı

import requests

API = "https://api.ticaretkopru.com/v1"
HEADERS = {"Authorization": "Bearer <firma_anahtari>"}

def ilan_aktar(urun):
    veri = {
        "tip": "satim",
        "baslik": urun["ad"],
        "gtip": urun["hs_kodu"],
        "miktar": urun["miktar_mt"],
        "birim": "MT",
        "teslim": urun["incoterm"],      # FOB / CIF / EXW
        "liman": urun["cikis_limani"],
        "suresiz": True,                 # ilan kendiliğinden kapanmaz
    }
    cevap = requests.post(f"{API}/ilanlar", json=veri, headers=HEADERS, timeout=20)
    cevap.raise_for_status()
    return cevap.json()["ilan_id"]

for urun in excel_satirlari("urunler.xlsx"):
    print(urun["ad"], "→", ilan_aktar(urun))
